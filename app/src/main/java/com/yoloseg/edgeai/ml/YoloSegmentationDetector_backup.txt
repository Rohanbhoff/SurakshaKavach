/**
 * ═══════════════════════════════════════════════════════════════════════════════
 *  YoloSegmentationDetector.kt
 *  TFLite Interpreter for YOLOv8s-Seg INT8 Quantized Model
 *
 *  Handles the complex dual-output tensor layout:
 *    Output 0: [1, 8400, 38] — Bounding boxes + class scores + mask coefficients
 *    Output 1: [1, 32, 160, 160] — Prototype mask matrices
 *
 *  Performs INT8 dequantization, NMS, and per-instance mask generation.
 * ═══════════════════════════════════════════════════════════════════════════════
 *
 *  Place at: app/src/main/java/com/yoloseg/edgeai/ml/YoloSegmentationDetector.kt
 */

package com.yoloseg.edgeai.ml

import android.content.Context
import android.graphics.Bitmap
import android.graphics.RectF
import android.util.Log
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.gpu.GpuDelegate
import org.tensorflow.lite.nnapi.NnApiDelegate
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.MappedByteBuffer
import java.nio.channels.FileChannel
import org.tensorflow.lite.DataType
import kotlin.math.exp
import kotlin.math.max
import kotlin.math.min

/**
 * A single segmentation detection result from the YOLOv8s-seg model.
 *
 * @property classId      Class index (0=Ladder, 1=Worker)
 * @property className    Human-readable class name
 * @property confidence   Detection confidence score [0.0, 1.0]
 * @property boundingBox  Bounding box in normalized coordinates [0.0, 1.0]
 * @property maskBitmap   160×160 binary segmentation mask bitmap (ARGB_8888)
 */
data class SegmentationResult(
    val classId: Int,
    val className: String,
    val confidence: Float,
    val boundingBox: RectF,
    val maskBitmap: Bitmap
)

/**
 * YOLOv8s-Seg TFLite Interpreter with INT8 dequantization support.
 *
 * Thread Safety: All inference calls are synchronized on an internal lock.
 * The TFLite Interpreter is NOT thread-safe by design.
 *
 * @param context   Android context for asset loading
 * @param modelPath Asset filename of the .tflite model (e.g., "best-int8.tflite")
 * @param useGpu    Enable GPU delegate acceleration (default: true)
 * @param useNnapi  Enable NNAPI delegate for NPU/DSP acceleration (default: false)
 */
class YoloSegmentationDetector(
    context: Context,
    modelPath: String = "best-int8.tflite",
    useGpu: Boolean = true,
    useNnapi: Boolean = false
) : AutoCloseable {

    // ── Model Configuration Constants ──
    companion object {
        const val INPUT_SIZE = 640
        const val NUM_CLASSES = 2
        const val NUM_ANCHORS = 8400        // YOLOv8 anchor grid: 80² + 40² + 20² = 8400
        const val NUM_MASK_COEFFS = 32       // Mask prototype coefficient count
        const val MASK_SIZE = 160            // Prototype mask resolution (640/4)
        const val VALUES_PER_ANCHOR = 4 + NUM_CLASSES + NUM_MASK_COEFFS  // = 38

        const val CONFIDENCE_THRESHOLD = 0.5f
        const val IOU_THRESHOLD = 0.45f
        const val MASK_THRESHOLD = 0.5f

        val CLASS_NAMES = arrayOf("Ladder", "Worker")
    }

    // ── Internal State ──
    private val interpreter: Interpreter
    private val gpuDelegate: GpuDelegate?
    private val nnapiDelegate: NnApiDelegate?
    private val lock = Any()
    
    // Last maximum confidence score found in the frame (for debugging on-screen)
    var lastMaxScore: Float = 0f

    // ── Pre-allocated Buffers (reused across frames) ──
    private val inputBuffer: ByteBuffer

    // ── Quantization Parameters ──
    private var inputScale: Float = 1f
    private var inputZeroPoint: Int = 0
    private var output0Scale: Float = 1f
    private var output0ZeroPoint: Int = 0
    private var output1Scale: Float = 1f
    private var output1ZeroPoint: Int = 0

    private var isInputQuantized: Boolean = false
    private var isOutput0Quantized: Boolean = false
    private var isOutput1Quantized: Boolean = false

    private var inputDataType: DataType = DataType.FLOAT32
    private var output0DataType: DataType = DataType.FLOAT32
    private var output1DataType: DataType = DataType.FLOAT32

    init {
        // Load the model file from assets
        val modelBuffer = loadModelFile(context, modelPath)

        // Probe tensor types and quantization parameters
        val probeInterpreter = Interpreter(modelBuffer, Interpreter.Options())
        
        val inputDetail = probeInterpreter.getInputTensor(0)
        inputDataType = inputDetail.dataType()
        isInputQuantized = (inputDataType == DataType.INT8 || inputDataType == DataType.UINT8)
        if (isInputQuantized) {
            val quant = inputDetail.quantizationParams()
            inputScale = quant.scale
            inputZeroPoint = quant.zeroPoint.toInt()
        }

        val out0Tensor = probeInterpreter.getOutputTensor(0)
        output0DataType = out0Tensor.dataType()
        isOutput0Quantized = (output0DataType == DataType.INT8 || output0DataType == DataType.UINT8)
        if (isOutput0Quantized) {
            val quant = out0Tensor.quantizationParams()
            output0Scale = quant.scale
            output0ZeroPoint = quant.zeroPoint.toInt()
        }

        val out1Tensor = probeInterpreter.getOutputTensor(1)
        output1DataType = out1Tensor.dataType()
        isOutput1Quantized = (output1DataType == DataType.INT8 || output1DataType == DataType.UINT8)
        if (isOutput1Quantized) {
            val quant = out1Tensor.quantizationParams()
            output1Scale = quant.scale
            output1ZeroPoint = quant.zeroPoint.toInt()
        }
        
        probeInterpreter.close()

        Log.i("YoloSeg", "Model probe:")
        Log.i("YoloSeg", "  Input: type=$inputDataType, quantized=$isInputQuantized, scale=$inputScale, zp=$inputZeroPoint")
        Log.i("YoloSeg", "  Output0: type=$output0DataType, quantized=$isOutput0Quantized, scale=$output0Scale, zp=$output0ZeroPoint")
        Log.i("YoloSeg", "  Output1: type=$output1DataType, quantized=$isOutput1Quantized, scale=$output1Scale, zp=$output1ZeroPoint")

        // Configure interpreter options with hardware delegates
        // GPU delegate does NOT support INT8 quantized models.
        val effectiveUseGpu = useGpu && !isInputQuantized

        val options = Interpreter.Options().apply {
            setNumThreads(4)
        }

        var gpuDel: GpuDelegate? = null
        var nnapiDel: NnApiDelegate? = null

        if (effectiveUseGpu) {
            try {
                gpuDel = GpuDelegate()
                options.addDelegate(gpuDel)
                Log.i("YoloSeg", "GPU delegate enabled (float model)")
            } catch (e: Exception) {
                gpuDel?.close()
                gpuDel = null
                Log.w("YoloSeg", "GPU delegate not available: ${e.message}")
            }
        } else if (useGpu && isInputQuantized) {
            Log.w("YoloSeg", "GPU delegate DISABLED — INT8 quantized models must run on CPU/NNAPI")
        }

        if (useNnapi && gpuDel == null) {
            try {
                nnapiDel = NnApiDelegate()
                options.addDelegate(nnapiDel)
                Log.i("YoloSeg", "NNAPI delegate enabled")
            } catch (e: Exception) {
                nnapiDel?.close()
                nnapiDel = null
                Log.w("YoloSeg", "NNAPI delegate not available: ${e.message}")
            }
        }

        gpuDelegate = gpuDel
        nnapiDelegate = nnapiDel
        interpreter = Interpreter(modelBuffer, options)

        Log.i("YoloSeg", "Interpreter created. Delegate: ${when {
            gpuDel != null -> "GPU"
            nnapiDel != null -> "NNAPI"
            else -> "CPU (4 threads)"
        }}")

        // Pre-allocate the input buffer
        val inputTensor = interpreter.getInputTensor(0)
        val inputBytes = inputTensor.numBytes()
        inputBuffer = ByteBuffer.allocateDirect(inputBytes).apply {
            order(ByteOrder.nativeOrder())
        }

        Log.i("YoloSeg", "Input tensor: shape=${inputTensor.shape().contentToString()}, dtype=$inputDataType, numBytes=$inputBytes")
    }

    /**
     * Expose probed model structure and quantization parameters for on-screen debugging.
     */
    val modelDetails: String
        get() {
            val inputShape = interpreter.getInputTensor(0).shape().contentToString()
            val out0Shape = interpreter.getOutputTensor(0).shape().contentToString()
            val out1Shape = interpreter.getOutputTensor(1).shape().contentToString()
            return "Input: $inputDataType, shape=$inputShape (scale=$inputScale, zp=$inputZeroPoint)\n" +
                   "Out0: $output0DataType, shape=$out0Shape (scale=$output0Scale, zp=$output0ZeroPoint)\n" +
                   "Out1: $output1DataType, shape=$out1Shape (scale=$output1Scale, zp=$output1ZeroPoint)"
        }

    /**
     * Run segmentation inference on a Bitmap.
     *
     * @param bitmap Input image (any size — will be resized to 640×640 internally)
     * @return List of [SegmentationResult] objects, sorted by confidence (descending)
     */
    fun detect(bitmap: Bitmap): List<SegmentationResult> = synchronized(lock) {
        // ── Step 1: Preprocess — Resize and fill input buffer ──
        val resized = Bitmap.createScaledBitmap(bitmap, INPUT_SIZE, INPUT_SIZE, true)
        fillInputBuffer(resized)
        if (resized != bitmap) resized.recycle()

        // ── Step 2: Allocate output buffers ──
        val output0Tensor = interpreter.getOutputTensor(0)
        val output1Tensor = interpreter.getOutputTensor(1)
        val output0Shape = output0Tensor.shape()  // [1, 8400, 38] or [1, 38, 8400]
        val output1Shape = output1Tensor.shape()  // [1, 32, 160, 160]

        Log.d("YoloSeg", "Output0 shape: ${output0Shape.contentToString()}, dtype=${output0Tensor.dataType()}")
        Log.d("YoloSeg", "Output1 shape: ${output1Shape.contentToString()}, dtype=${output1Tensor.dataType()}")

        // Determine if output0 is [1, anchors, values] or [1, values, anchors]
        val isTransposed = output0Shape[1] == VALUES_PER_ANCHOR && output0Shape[2] == NUM_ANCHORS
        Log.d("YoloSeg", "Output0 layout: ${if (isTransposed) "TRANSPOSED [1,38,8400]" else "STANDARD [1,8400,38]"}")

        val output0: Any
        val output1: Any

        if (isOutput0Quantized) {
            output0 = Array(output0Shape[0]) { Array(output0Shape[1]) { ByteArray(output0Shape[2]) } }
        } else {
            output0 = Array(output0Shape[0]) { Array(output0Shape[1]) { FloatArray(output0Shape[2]) } }
        }

        if (isOutput1Quantized) {
            output1 = Array(output1Shape[0]) { Array(output1Shape[1]) { Array(output1Shape[2]) { ByteArray(output1Shape[3]) } } }
        } else {
            output1 = Array(output1Shape[0]) { Array(output1Shape[1]) { Array(output1Shape[2]) { FloatArray(output1Shape[3]) } } }
        }

        val outputMap = mapOf(0 to output0, 1 to output1)

        // ── Step 3: Run inference ──
        interpreter.runForMultipleInputsOutputs(arrayOf(inputBuffer), outputMap)

        // ── Step 4: Dequantize and parse output tensors ──
        val detections = parseDetections(output0, isTransposed)
        val protoMasks = parseProtoMasks(output1)

        // ── Step 5: Apply Non-Maximum Suppression ──
        val nmsResults = applyNms(detections)

        Log.i("YoloSeg", "Post-NMS: ${nmsResults.size} detections (from ${detections.size} pre-NMS)")

        // ── Step 6: Generate per-instance masks ──
        return@synchronized nmsResults.map { det ->
            val mask = generateInstanceMask(det.maskCoeffs, protoMasks, det.box)
            SegmentationResult(
                classId = det.classId,
                className = CLASS_NAMES.getOrElse(det.classId) { "Unknown" },
                confidence = det.confidence,
                boundingBox = det.box,
                maskBitmap = mask
            )
        }.sortedByDescending { it.confidence }
    }

    // ═══════════════════════════════════════════════════════════════════════════
    //  Input Preprocessing
    // ═══════════════════════════════════════════════════════════════════════════

    /**
     * Fill the input ByteBuffer from a 640×640 Bitmap.
     * Applies INT8 quantization if the model expects quantized input.
     */
    private fun fillInputBuffer(bitmap: Bitmap) {
        inputBuffer.rewind()
        val pixels = IntArray(INPUT_SIZE * INPUT_SIZE)
        bitmap.getPixels(pixels, 0, INPUT_SIZE, 0, 0, INPUT_SIZE, INPUT_SIZE)

        for (pixel in pixels) {
            val r = (pixel shr 16) and 0xFF
            val g = (pixel shr 8) and 0xFF
            val b = pixel and 0xFF

            if (isInputQuantized) {
                // Normalize [0,255] → [0.0,1.0] → quantize to INT8 or UINT8
                // quantized_val = float_val / scale + zero_point
                val minQuant = if (inputDataType == DataType.UINT8) 0 else -128
                val maxQuant = if (inputDataType == DataType.UINT8) 255 else 127

                inputBuffer.put(((r / 255.0f / inputScale) + inputZeroPoint).toInt().coerceIn(minQuant, maxQuant).toByte())
                inputBuffer.put(((g / 255.0f / inputScale) + inputZeroPoint).toInt().coerceIn(minQuant, maxQuant).toByte())
                inputBuffer.put(((b / 255.0f / inputScale) + inputZeroPoint).toInt().coerceIn(minQuant, maxQuant).toByte())
            } else {
                // Float32 input — normalize to [0.0, 1.0]
                inputBuffer.putFloat(r / 255.0f)
                inputBuffer.putFloat(g / 255.0f)
                inputBuffer.putFloat(b / 255.0f)
            }
        }
    }

    // ═══════════════════════════════════════════════════════════════════════════
    //  Output Tensor Parsing with INT8 Dequantization
    // ═══════════════════════════════════════════════════════════════════════════

    /**
     * Internal detection candidate before NMS.
     */
    private data class RawDetection(
        val classId: Int,
        val confidence: Float,
        val box: RectF,
        val maskCoeffs: FloatArray
    )

    /**
     * Parse Output Tensor 0 — bounding boxes, class scores, mask coefficients.
     *
     * Handles both layouts:
     *   Standard:   [1, 8400, 38] — each row is one anchor
     *   Transposed: [1, 38, 8400] — each column is one anchor
     *
     * INT8 Dequantization: float_val = (int8_val - zeroPoint) * scale
     */
    private fun parseDetections(output0: Any, isTransposed: Boolean): List<RawDetection> {
        val detections = mutableListOf<RawDetection>()
        var maxRawScore = -9999f    // Before sigmoid
        var maxSigScore = -9999f    // After sigmoid
        var maxScoreClass = 0

        for (a in 0 until NUM_ANCHORS) {
            val values = FloatArray(VALUES_PER_ANCHOR)

            for (v in 0 until VALUES_PER_ANCHOR) {
                val rawVal: Float = if (isOutput0Quantized) {
                    @Suppress("UNCHECKED_CAST")
                    val arr = output0 as Array<Array<ByteArray>>
                    val byteVal = if (isTransposed) arr[0][v][a] else arr[0][a][v]
                    // ── Dequantization ──
                    // CRITICAL FIX: Handle both INT8 (signed, -128..127) and
                    // UINT8 (unsigned, 0..255) tensor data types correctly.
                    // Kotlin Byte is always signed. For UINT8, we must mask
                    // with 0xFF to get the unsigned integer value.
                    val intVal = if (output0DataType == DataType.UINT8) {
                        byteVal.toInt() and 0xFF   // Unsigned: 0..255
                    } else {
                        byteVal.toInt()            // Signed: -128..127
                    }
                    (intVal - output0ZeroPoint) * output0Scale
                } else {
                    @Suppress("UNCHECKED_CAST")
                    val arr = output0 as Array<Array<FloatArray>>
                    if (isTransposed) arr[0][v][a] else arr[0][a][v]
                }
                values[v] = rawVal
            }

            // Parse: [cx, cy, w, h, cls0_score, cls1_score, 32×mask_coeffs]
            val cx = values[0]
            val cy = values[1]
            val w = values[2]
            val h = values[3]

            // Find best class — apply sigmoid to raw logits first.
            // YOLOv8 outputs raw logits, NOT probabilities.
            var bestClassId = 0
            var bestRawScore = values[4]
            for (c in 1 until NUM_CLASSES) {
                if (values[4 + c] > bestRawScore) {
                    bestRawScore = values[4 + c]
                    bestClassId = c
                }
            }

            // YOLOv8 TFLite export already applies sigmoid to class scores.
            // DO NOT apply it again, or 0.0 becomes 0.5, creating false positives!
            val bestScore = bestRawScore

            if (bestRawScore > maxRawScore) {
                maxRawScore = bestRawScore
                maxSigScore = bestScore
                maxScoreClass = bestClassId
            }

            // Apply confidence threshold (on sigmoid-activated score)
            if (bestScore < CONFIDENCE_THRESHOLD) continue

            // Convert center-format to corner-format (normalized [0,1])
            val x1 = (cx - w / 2f) / INPUT_SIZE
            val y1 = (cy - h / 2f) / INPUT_SIZE
            val x2 = (cx + w / 2f) / INPUT_SIZE
            val y2 = (cy + h / 2f) / INPUT_SIZE

            // Clamp to valid range
            val box = RectF(
                max(0f, x1), max(0f, y1),
                min(1f, x2), min(1f, y2)
            )

            // Extract 32 mask coefficients
            val maskCoeffs = FloatArray(NUM_MASK_COEFFS)
            System.arraycopy(values, 4 + NUM_CLASSES, maskCoeffs, 0, NUM_MASK_COEFFS)

            detections.add(RawDetection(bestClassId, bestScore, box, maskCoeffs))
        }

        Log.i("YoloSeg", "Max raw logit: $maxRawScore → sigmoid: $maxSigScore " +
            "for class ${CLASS_NAMES.getOrElse(maxScoreClass) { maxScoreClass.toString() }} " +
            "(Threshold: $CONFIDENCE_THRESHOLD). Detections passing: ${detections.size}")
        lastMaxScore = maxSigScore
        return detections
    }

    /**
     * Parse Output Tensor 1 — 32 prototype masks at 160×160 resolution.
     *
     * INT8 Dequantization: float_val = (int8_val - zeroPoint) * scale
     *
     * @return 2D array [32][160*160] of float prototype mask values
     */
    private fun parseProtoMasks(output1: Any): Array<FloatArray> {
        val protos = Array(NUM_MASK_COEFFS) { FloatArray(MASK_SIZE * MASK_SIZE) }

        for (p in 0 until NUM_MASK_COEFFS) {
            for (y in 0 until MASK_SIZE) {
                for (x in 0 until MASK_SIZE) {
                    val idx = y * MASK_SIZE + x
                    protos[p][idx] = if (isOutput1Quantized) {
                        @Suppress("UNCHECKED_CAST")
                        val arr = output1 as Array<Array<Array<ByteArray>>>
                        val byteVal = arr[0][p][y][x]
                        // ── Dequantization (signed/unsigned aware) ──
                        val intVal = if (output1DataType == DataType.UINT8) {
                            byteVal.toInt() and 0xFF
                        } else {
                            byteVal.toInt()
                        }
                        (intVal - output1ZeroPoint) * output1Scale
                    } else {
                        @Suppress("UNCHECKED_CAST")
                        val arr = output1 as Array<Array<Array<FloatArray>>>
                        arr[0][p][y][x]
                    }
                }
            }
        }

        return protos
    }

    // ═══════════════════════════════════════════════════════════════════════════
    //  Non-Maximum Suppression (NMS)
    // ═══════════════════════════════════════════════════════════════════════════

    /**
     * Apply class-aware Non-Maximum Suppression.
     * Filters overlapping detections by IoU threshold.
     */
    private fun applyNms(detections: List<RawDetection>): List<RawDetection> {
        if (detections.isEmpty()) return emptyList()

        // Sort by confidence descending
        val sorted = detections.sortedByDescending { it.confidence }
        val selected = mutableListOf<RawDetection>()
        val suppressed = BooleanArray(sorted.size)

        for (i in sorted.indices) {
            if (suppressed[i]) continue
            selected.add(sorted[i])

            for (j in i + 1 until sorted.size) {
                if (suppressed[j]) continue
                // Only suppress within the same class
                if (sorted[i].classId != sorted[j].classId) continue

                val iou = computeIou(sorted[i].box, sorted[j].box)
                if (iou > IOU_THRESHOLD) {
                    suppressed[j] = true
                }
            }
        }

        return selected
    }

    /**
     * Compute Intersection over Union (IoU) between two bounding boxes.
     */
    private fun computeIou(a: RectF, b: RectF): Float {
        val interLeft = max(a.left, b.left)
        val interTop = max(a.top, b.top)
        val interRight = min(a.right, b.right)
        val interBottom = min(a.bottom, b.bottom)

        val interArea = max(0f, interRight - interLeft) * max(0f, interBottom - interTop)
        val aArea = (a.right - a.left) * (a.bottom - a.top)
        val bArea = (b.right - b.left) * (b.bottom - b.top)
        val unionArea = aArea + bArea - interArea

        return if (unionArea > 0f) interArea / unionArea else 0f
    }

    // ═══════════════════════════════════════════════════════════════════════════
    //  Per-Instance Mask Generation
    // ═══════════════════════════════════════════════════════════════════════════

    /**
     * Generate a 160×160 instance segmentation mask by combining
     * the detection's 32 mask coefficients with the 32 prototype masks.
     *
     * mask = sigmoid(Σ(coeff_i × proto_i)) for each pixel
     * Then crop to the bounding box region and threshold at 0.5.
     */
    private fun generateInstanceMask(
        coeffs: FloatArray,
        protos: Array<FloatArray>,
        box: RectF
    ): Bitmap {
        val maskPixels = IntArray(MASK_SIZE * MASK_SIZE)

        // Compute bounding box in mask coordinates (160×160)
        val boxLeft = (box.left * MASK_SIZE).toInt().coerceIn(0, MASK_SIZE - 1)
        val boxTop = (box.top * MASK_SIZE).toInt().coerceIn(0, MASK_SIZE - 1)
        val boxRight = (box.right * MASK_SIZE).toInt().coerceIn(0, MASK_SIZE - 1)
        val boxBottom = (box.bottom * MASK_SIZE).toInt().coerceIn(0, MASK_SIZE - 1)

        for (y in 0 until MASK_SIZE) {
            for (x in 0 until MASK_SIZE) {
                val idx = y * MASK_SIZE + x

                // Only compute mask within the bounding box region
                if (x < boxLeft || x > boxRight || y < boxTop || y > boxBottom) {
                    maskPixels[idx] = 0x00000000  // Transparent
                    continue
                }

                // Dot product: Σ(coeff_i × proto_i[pixel])
                var sum = 0f
                for (p in 0 until NUM_MASK_COEFFS) {
                    sum += coeffs[p] * protos[p][idx]
                }

                // Sigmoid activation
                val sigVal = sigmoid(sum)

                // Threshold and assign color
                if (sigVal > MASK_THRESHOLD) {
                    // Semi-transparent mask — alpha = 128 (50% opacity)
                    maskPixels[idx] = 0x80_00FF00.toInt()  // Green tint (will be recolored per-class)
                } else {
                    maskPixels[idx] = 0x00000000  // Transparent
                }
            }
        }

        return Bitmap.createBitmap(maskPixels, MASK_SIZE, MASK_SIZE, Bitmap.Config.ARGB_8888)
    }

    /**
     * Sigmoid activation function: σ(x) = 1 / (1 + e^(-x))
     */
    private fun sigmoid(x: Float): Float {
        return 1.0f / (1.0f + exp(-x))
    }

    // ═══════════════════════════════════════════════════════════════════════════
    //  Utility Functions
    // ═══════════════════════════════════════════════════════════════════════════

    /**
     * Load a TFLite model file from the app's assets directory.
     * Uses memory-mapped file I/O for zero-copy loading.
     */
    private fun loadModelFile(context: Context, modelPath: String): MappedByteBuffer {
        val assetFileDescriptor = context.assets.openFd(modelPath)
        val fileInputStream = FileInputStream(assetFileDescriptor.fileDescriptor)
        val fileChannel = fileInputStream.channel
        val startOffset = assetFileDescriptor.startOffset
        val declaredLength = assetFileDescriptor.declaredLength
        return fileChannel.map(FileChannel.MapMode.READ_ONLY, startOffset, declaredLength)
    }

    /**
     * Release all resources — must be called when the detector is no longer needed.
     */
    override fun close() {
        synchronized(lock) {
            interpreter.close()
            gpuDelegate?.close()
            nnapiDelegate?.close()
        }
    }
}
